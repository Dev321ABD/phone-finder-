package com.example.phonefinder

import android.Manifest
import android.bluetooth.BluetoothAdapter
import android.bluetooth.BluetoothDevice
import android.bluetooth.le.*
import android.content.pm.PackageManager
import android.media.AudioManager
import android.media.ToneGenerator
import android.os.*
import android.widget.*
import androidx.activity.ComponentActivity
import androidx.core.app.ActivityCompat

class MainActivity : ComponentActivity() {

    private lateinit var scanner: BluetoothLeScanner
    private lateinit var status: TextView
    private lateinit var signal: TextView
    private lateinit var list: LinearLayout

    private var targetAddress: String? = null
    private var scanning = false

    private val handler = Handler(Looper.getMainLooper())
    private var lastBeep = 0L

    private val tone = ToneGenerator(
        AudioManager.STREAM_NOTIFICATION,
        100
    )

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        createUI()
        requestPermissions()
    }

    private fun createUI() {

        val root = LinearLayout(this)
        root.orientation = LinearLayout.VERTICAL
        root.setPadding(25, 25, 25, 25)

        val title = TextView(this)
        title.text = "?? Android Phone Finder"
        title.textSize = 25f

        val scanButton = Button(this)
        scanButton.text = "SCAN"

        status = TextView(this)
        status.text = "Bluetooth ON karo"
        status.textSize = 18f

        signal = TextView(this)
        signal.text = "Target: None"
        signal.textSize = 24f

        list = LinearLayout(this)
        list.orientation = LinearLayout.VERTICAL

        root.addView(title)
        root.addView(scanButton)
        root.addView(status)
        root.addView(signal)
        root.addView(list)

        setContentView(root)

        scanButton.setOnClickListener {
            startScan()
        }
    }

    private fun requestPermissions() {

        if (Build.VERSION.SDK_INT >= 31) {

            ActivityCompat.requestPermissions(
                this,
                arrayOf(
                    Manifest.permission.BLUETOOTH_SCAN,
                    Manifest.permission.BLUETOOTH_CONNECT
                ),
                100
            )
        }
    }

    private fun startScan() {

        val adapter = BluetoothAdapter.getDefaultAdapter()

        if (adapter == null) {
            status.text = "Bluetooth available nahi hai"
            return
        }

        if (!adapter.isEnabled) {
            status.text = "Bluetooth ON karo"
            return
        }

        if (
            Build.VERSION.SDK_INT >= 31 &&
            ActivityCompat.checkSelfPermission(
                this,
                Manifest.permission.BLUETOOTH_SCAN
            ) != PackageManager.PERMISSION_GRANTED
        ) {
            requestPermissions()
            return
        }

        scanner = adapter.bluetoothLeScanner

        list.removeAllViews()
        status.text = "?? Scanning..."

        scanner.startScan(scanCallback)

        scanning = true

        handler.postDelayed({
            stopScan()
        }, 10000)
    }

    private fun stopScan() {

        if (!scanning) return

        try {
            scanner.stopScan(scanCallback)
        } catch (_: Exception) {
        }

        scanning = false
        status.text = "Device select karo"
    }

    private val scanCallback = object : ScanCallback() {

        override fun onScanResult(
            callbackType: Int,
            result: ScanResult
        ) {

            val device = result.device
            val rssi = result.rssi

            val name = try {
                device.name ?: "Unknown BLE device"
            } catch (_: SecurityException) {
                "Unknown BLE device"
            }

            runOnUiThread {

                /*
                 * Same device baar-baar list mein add na ho
                 */
                if (
                    list.findViewWithTag<Button>(device.address) == null
                ) {

                    val button = Button(this@MainActivity)

                    button.tag = device.address

                    button.text =
                        "$name\n$rssi dBm\n${device.address}"

                    button.setOnClickListener {

                        targetAddress = device.address

                        status.text =
                            "?? Target selected\n$name"

                        startTracking()
                    }

                    list.addView(button)
                }
            }
        }

        override fun onScanFailed(errorCode: Int) {

            runOnUiThread {
                status.text = "Scan failed: $errorCode"
            }
        }
    }

    private fun startTracking() {

        val target = targetAddress ?: return

        status.text = "?? Tracking..."

        scanner.startScan(
            object : ScanCallback() {

                override fun onScanResult(
                    callbackType: Int,
                    result: ScanResult
                ) {

                    if (result.device.address == target) {

                        val rssi = result.rssi

                        runOnUiThread {
                            updateSignal(rssi)
                        }
                    }
                }
            }
        )
    }

    private fun updateSignal(rssi: Int) {

        val level = when {

            rssi >= -45 -> {
                "?? BAHUT PAAS"
            }

            rssi >= -60 -> {
                "?? PAAS"
            }

            rssi >= -75 -> {
                "?? MEDIUM"
            }

            else -> {
                "?? DOOR"
            }
        }

        signal.text =
            "?? $rssi dBm\n\n$level"

        beepAccordingToRSSI(rssi)
    }

    private fun beepAccordingToRSSI(rssi: Int) {

        val now = System.currentTimeMillis()

        /*
         * Strong signal = faster beep
         */

        val interval = when {

            rssi >= -45 -> 250L

            rssi >= -60 -> 500L

            rssi >= -75 -> 1000L

            else -> 1800L
        }

        if (now - lastBeep >= interval) {

            tone.startTone(
                ToneGenerator.TONE_PROP_BEEP,
                100
            )

            lastBeep = now
        }
    }

    override fun onDestroy() {

        super.onDestroy()

        try {
            tone.release()
            scanner.stopScan(scanCallback)
        } catch (_: Exception) {
        }
    }
}
